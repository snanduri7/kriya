"""Discriminating check for presence_penalty (no effect at temperature 0 on
either adapter): sampled decoding (temperature 0.7, fixed seed), values 0 /
2.0 / 10.0, with repeat_penalty as the control, each twice."""
import hashlib, json, urllib.request

MODEL = "qwen3-coder:30b-kriya-e52213655394"
MSG = [{"role": "user", "content": "Write the word cat forty times, separated by spaces, then the word dog."}]


def native(**options):
    body = {"model": MODEL, "messages": MSG, "stream": False,
            "options": {"num_predict": 120, "temperature": 0.7, "seed": 7, **options}}
    req = urllib.request.Request("http://localhost:11434/api/chat", json.dumps(body).encode(),
                                 {"Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=600) as resp:
        return hashlib.sha256(json.load(resp)["message"]["content"].encode()).hexdigest()[:12]


out = {f"presence_penalty={v}": [native(presence_penalty=v) for _ in range(2)] for v in (0.0, 2.0, 10.0)}
out.update({f"repeat_penalty={v}": [native(repeat_penalty=v) for _ in range(2)] for v in (1.0, 2.0)})
print(json.dumps(out, indent=1))
