"""PROVIDER-CONTRACT-001A L2/L3 (live): a local proxy in front of the real
Ollama fails /api/ps on demand and passes everything else through unchanged.
The v4 production config (native adapter) is loaded as approved and pointed
at the proxy in memory. L2: the first /api/ps of each call fails once ->
observed anyway (one bounded retry), and every later call observes again.
L3: /api/ps always fails -> strict production refuses SERVED_CONTEXT_UNOBSERVABLE.
Run from the v4 workspace: python l2_l3_probe.py <v4 config>"""
import asyncio, json, sys, threading, urllib.request
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

from kriya.config.config import load_config
from kriya.core.llm import LLMClient

UPSTREAM = "http://localhost:11434"
state = {"fail_next_ps": 0, "fail_all_ps": False, "ps_seen": 0, "ps_failed": 0}


class Proxy(BaseHTTPRequestHandler):
    def log_message(self, *args):
        pass

    def _forward(self, method):
        if self.path.startswith("/api/ps"):
            state["ps_seen"] += 1
            if state["fail_all_ps"] or state["fail_next_ps"] > 0:
                state["fail_next_ps"] = max(0, state["fail_next_ps"] - 1)
                state["ps_failed"] += 1
                self.send_response(503); self.send_header("Content-Length", "0"); self.end_headers()
                return
        body = self.rfile.read(int(self.headers.get("Content-Length") or 0)) if method == "POST" else None
        req = urllib.request.Request(UPSTREAM + self.path, body, {"Content-Type": "application/json"}, method=method)
        try:
            with urllib.request.urlopen(req, timeout=900) as resp:
                data, status = resp.read(), resp.status
        except urllib.error.HTTPError as error:
            data, status = error.read(), error.code
        self.send_response(status); self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(data))); self.end_headers(); self.wfile.write(data)

    def do_GET(self):
        self._forward("GET")

    def do_POST(self):
        self._forward("POST")


server = ThreadingHTTPServer(("127.0.0.1", 0), Proxy)
threading.Thread(target=server.serve_forever, daemon=True).start()
url = f"http://127.0.0.1:{server.server_port}/v1"
cfg = load_config(sys.argv[1])
cfg.llm.base_url = url
for fallback in cfg.llm_chain:
    fallback.base_url = url


async def call(label):
    llm = LLMClient(cfg)
    try:
        r = await llm.complete_result("Answer tersely.", "Reply with OK.", max_tokens_override=8)
    finally:
        await llm.aclose()
    c = r.protocol["provider_contract"]
    print(json.dumps({"case": label, "status": r.status.value, "runtime_exact": r.runtime_fingerprint_exact,
                      "observation": c.get("served_context_observation"),
                      "served_after_call": c.get("served_context_window_after_call"),
                      "violation": (c.get("violation") or {}).get("reason_code") if isinstance(c.get("violation"), dict)
                      else getattr(r.error, "reason_code", None),
                      "ps_seen": state["ps_seen"], "ps_failed": state["ps_failed"]}), flush=True)


async def main():
    await call("baseline")
    state["fail_next_ps"] = 1
    await call("L2_one_transient_failure")
    await call("L2_next_call_observes_again")
    state["fail_all_ps"] = True
    await call("L3_persistent_failure_production")
    state["fail_all_ps"] = False
    await call("L3_recovered")


asyncio.run(main())
server.shutdown()
