"""Deterministic transport measurements: SDK retries, timeout default, proxy inheritance, streaming resend."""
import asyncio, json, os, sys, threading, socketserver
from http.server import BaseHTTPRequestHandler, HTTPServer
sys.path.insert(0, ".")
REQS = []

class H(BaseHTTPRequestHandler):
    mode = "500"
    def log_message(self, *a): pass
    def do_POST(self):
        body = json.loads(self.rfile.read(int(self.headers["Content-Length"])))
        REQS.append({"path": self.path, "stream": body.get("stream"), "stream_options": body.get("stream_options")})
        if H.mode == "500":
            self.send_response(500); self.end_headers(); self.wfile.write(b'{"error":"boom"}')
        elif H.mode == "stream_400":
            if body.get("stream_options"):
                self.send_response(500); self.end_headers(); self.wfile.write(b'{"error":"transient"}')
            else:
                self.send_response(200); self.send_header("Content-Type", "text/event-stream"); self.end_headers()
                self.wfile.write(b'data: {"id":"x","object":"chat.completion.chunk","created":0,"model":"m","choices":[{"index":0,"delta":{"content":"hi"},"finish_reason":"stop"}]}\n\ndata: [DONE]\n\n')

srv = HTTPServer(("127.0.0.1", 0), H); port = srv.server_port
threading.Thread(target=srv.serve_forever, daemon=True).start()

class Proxy(socketserver.BaseRequestHandler):
    hits = 0
    def handle(self):
        Proxy.hits += 1; self.request.recv(4096); self.request.sendall(b"HTTP/1.1 502 Bad Gateway\r\nContent-Length: 0\r\n\r\n")
proxy = socketserver.TCPServer(("127.0.0.1", 0), Proxy); pport = proxy.server_address[1]
threading.Thread(target=proxy.serve_forever, daemon=True).start()

from kriya.config import AppConfig
from kriya.core.llm import LLMClient
from kriya.core.inference_runtime import ChatRequest
from kriya.core import model_runtime

async def main():
    cfg = AppConfig(); cfg.llm.base_url = f"http://127.0.0.1:{port}/v1"; cfg.llm.model = "m"
    llm = LLMClient(cfg)
    c = llm.client
    print("P0-5 sdk max_retries:", c.max_retries, " timeout:", c.timeout)
    t = c._client
    print("P0-6 httpx trust_env:", getattr(t, "_trust_env", None))
    adapter = llm._runtime("m")
    H.mode = "500"; REQS.clear()
    try:
        await adapter.complete(c, ChatRequest(model="m", messages=[{"role":"user","content":"x"}], temperature=0, max_tokens=4))
    except Exception as e:
        print("P0-5 one non-stream call ->", len(REQS), "HTTP requests; error", type(e).__name__)
    H.mode = "stream_400"; REQS.clear()
    r = await adapter.complete(llm.client.with_options(max_retries=0), ChatRequest(model="m", messages=[{"role":"user","content":"x"}], temperature=0, max_tokens=4, stream_callback=lambda t: None))
    print("P0-7 one streaming call (SDK retries=0) ->", len(REQS), "HTTP requests:", REQS, "content", repr(r.content))
    os.environ["HTTP_PROXY"] = os.environ["http_proxy"] = f"http://127.0.0.1:{pport}"
    os.environ.pop("NO_PROXY", None); os.environ.pop("no_proxy", None)
    llm2 = LLMClient(cfg); H.mode = "500"; REQS.clear(); Proxy.hits = 0
    try:
        await adapter.complete(llm2.client.with_options(max_retries=0), ChatRequest(model="m", messages=[{"role":"user","content":"x"}], temperature=0, max_tokens=4))
    except Exception as e:
        print("P0-6 with HTTP_PROXY set: proxy hits", Proxy.hits, "direct server hits", len(REQS), type(e).__name__)
asyncio.run(main())
