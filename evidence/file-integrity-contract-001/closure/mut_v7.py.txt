import os, shutil, subprocess, sys
P = "kriya/core/model_qualification.py"
M = [
 ("drop exactness", P, '        payload_exact = parsed.kind == "file" and content == FULL_FILE_SOURCE\n', '        payload_exact = parsed.kind == "file"\n'),
 ("bypass protocol parsing", P, '    parsed = (parse_structured(raw, "slug.py", patch_allowed=False) if protocol == STRUCTURED\n              else DeveloperAgent.parse_file_payload(raw, "slug.py"))\n    content = (parsed.content or "") if parsed.kind == "file" else ""\n    # Model output is never executed on the host: the check is structural.\n    defines = False',
  '    parsed = DeveloperAgent.parse_file_payload(raw, "slug.py")\n    content = (parsed.content or "") if parsed.kind == "file" else ""\n    # Model output is never executed on the host: the check is structural.\n    defines = False'),
 ("accept dropped fence", P, '        payload_exact = parsed.kind == "file" and content == FULL_FILE_SOURCE\n',
  '        payload_exact = parsed.kind == "file" and content.replace("    ```python\\n", "").replace("    ```\\n", "") == FULL_FILE_SOURCE.replace("    ```python\\n", "").replace("    ```\\n", "")\n'),
 ("accept trailing prose", "kriya/agents/response_protocol.py", '            elif line.strip():\n                return _fail(INVALID_EDIT_PROTOCOL, "text after the last protocol block")',
  '            elif False:\n                return _fail(INVALID_EDIT_PROTOCOL, "text after the last protocol block")'),
]
for name, path, old, new in M:
    src = open(path).read()
    if src.count(old) != 1:
        print("NOT APPLICABLE", name, src.count(old)); continue
    shutil.copy2(path, path + ".bak")
    try:
        open(path, "w").write(src.replace(old, new))
        r = subprocess.run([sys.executable, "-m", "pytest", "-q", "-x", "-p", "no:cacheprovider", "tests/test_file_integrity_contract_001.py",
                            "-k", "v7 or v6_records"], capture_output=True, text=True)
        print("KILLED" if r.returncode else "SURVIVED", name, r.stdout.strip().splitlines()[-1])
    finally:
        shutil.move(path + ".bak", path)
