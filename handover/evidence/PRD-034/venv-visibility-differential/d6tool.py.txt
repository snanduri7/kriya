#!/usr/bin/env python3
"""Read-only observation tooling for the PRD-034 venv-visibility differential (owner ruling 2026-11-04, option 1).

  d6tool.py make-shim DIR REAL_DOCKER LOGDIR  -> writes DIR/docker (logs every call, delegates unchanged; never alters argv, adds no wait)
  d6tool.py shim-run REAL LOGDIR ARGS...      -> the shim body
  d6tool.py sample LOGDIR T0 JSON_PATHS TAG   -> detached host-visibility sampler (read-only lstat) at 50ms..5s after container exit
  d6tool.py watch ROOT OUT                    -> continuous (20 ms) transition log of every <ws>/.kriya/venv under ROOT
Nothing here writes into a workspace, waits inside the product, or changes a docker argument.
"""
import json
import os
import signal
import subprocess
import sys
import threading
import time

SCHEDULE = (0.05, 0.1, 0.25, 0.5, 1.0, 2.0, 5.0)  # seconds after container exit; the 0 ms sample is taken inside the shim
CONTAINER_WS = "/kriya/workspace"


def now():
    return time.time()


def lst(path):
    try:
        st = os.lstat(path)
    except OSError as error:
        return {"p": path, "x": False, "err": type(error).__name__ + ":" + str(error.errno)}
    info = {"p": path, "x": True, "mode": oct(st.st_mode & 0o7777), "type": "lnk" if os.path.islink(path) else ("dir" if os.path.isdir(path) else "file"),
            "uid": st.st_uid, "gid": st.st_gid, "ino": st.st_ino}
    if info["type"] == "dir":
        try:
            info["n"] = len(os.listdir(path))
        except OSError as error:
            info["n"] = "err:" + str(error.errno)
    return info


def write_jsonl(path, row):
    with open(path, "a", encoding="utf-8") as handle:
        handle.write(json.dumps(row, sort_keys=True) + "\n")


def mounts_of(argv):
    found = []
    for i, token in enumerate(argv):
        spec = None
        if token in ("-v", "--volume") and i + 1 < len(argv):
            spec = argv[i + 1]
        elif token.startswith("--volume="):
            spec = token.split("=", 1)[1]
        elif token == "--mount" and i + 1 < len(argv):
            fields = dict(part.split("=", 1) for part in argv[i + 1].split(",") if "=" in part)
            found.append({"src": fields.get("source") or fields.get("src"), "dst": fields.get("target") or fields.get("dst"), "ro": "readonly" in argv[i + 1]})
            continue
        if spec:
            parts = spec.split(":")
            found.append({"src": parts[0], "dst": parts[1] if len(parts) > 1 else None, "ro": len(parts) > 2 and "ro" in parts[2].split(",")})
    return found


def watch_paths(mounts):
    """Host paths whose visibility matters for the venv mechanism, derived from the mounts of one docker run."""
    paths = []
    for mount in mounts:
        src, dst = mount["src"], mount["dst"]
        if not src or not os.path.isabs(src):
            continue
        if dst == CONTAINER_WS:
            paths += [src, src + "/.kriya", src + "/.kriya/venv", src + "/.kriya/venv/bin", src + "/.kriya/venv/bin/python", src + "/.kriya/venv/pyvenv.cfg"]
        elif dst == CONTAINER_WS + "/.kriya/venv":
            paths += [src, src + "/bin", src + "/bin/python", src + "/pyvenv.cfg"]
        elif dst == CONTAINER_WS + "/.kriya":
            paths += [src, src + "/venv", src + "/venv/bin/python"]
    seen, unique = set(), []
    for path in paths:
        if path not in seen:
            seen.add(path)
            unique.append(path)
    return unique


def sidecar(real, name, logdir, tag, stop):
    """Concurrent, read-only look INSIDE the running container (docker exec), at most ~7/s. Nothing is waited on by the product."""
    probe = ("for p in {w}/.kriya {w}/.kriya/venv {w}/.kriya/venv/bin {w}/.kriya/venv/bin/python {w}/.kriya/venv/pyvenv.cfg; do "
             "if [ -L \"$p\" ]; then echo \"L:$p->$(readlink \"$p\")\"; elif [ -e \"$p\" ]; then echo \"Y:$p\"; else echo \"N:$p\"; fi; done").format(w=CONTAINER_WS)
    last = None
    while not stop.is_set():
        started = now()
        result = subprocess.run([real, "exec", name, "sh", "-c", probe], capture_output=True, text=True, timeout=20, check=False)
        out = (result.stdout or "").strip().splitlines() if result.returncode == 0 else ["exec-rc=%d" % result.returncode]
        if out != last:
            write_jsonl(os.path.join(logdir, "inside.jsonl"), {"tag": tag, "t": started, "container": name, "inside": out})
            last = out
        stop.wait(0.05)


def shim_run(real, logdir, argv):
    started = now()
    verb = next((a for a in argv if not a.startswith("-")), "")
    is_run = bool(argv) and argv[0] == "run"
    joined = " ".join(argv)
    name = None
    for i, token in enumerate(argv):
        if token == "--name" and i + 1 < len(argv):
            name = argv[i + 1]
        elif token.startswith("--name="):
            name = token.split("=", 1)[1]
    tag = "%d-%d" % (os.getpid(), int(started * 1000))
    relevant = is_run and (".kriya" in joined or "venv" in joined)
    mounts = mounts_of(argv) if is_run else []
    row = {"tag": tag, "t0": started, "verb": verb, "argv": argv if is_run else argv[:6], "mounts": mounts, "pid": os.getpid()}
    if is_run:
        row["pre"] = [lst(p) for p in watch_paths(mounts)]
    child = subprocess.Popen([real] + argv)
    for signum in (signal.SIGTERM, signal.SIGINT, signal.SIGHUP):
        signal.signal(signum, lambda s, _f, c=child: c.send_signal(s))
    stop = threading.Event()
    thread = None
    if relevant and name:
        thread = threading.Thread(target=sidecar, args=(real, name, logdir, tag, stop), daemon=True)
        thread.start()
    code = child.wait()
    ended = now()
    stop.set()
    row.update(t1=ended, rc=code, wall=round(ended - started, 3))
    if is_run:
        row["post0"] = [lst(p) for p in watch_paths(mounts)]  # taken in-process, microseconds after the client returned
        row["post0_t"] = now()
    write_jsonl(os.path.join(logdir, "docker-calls.jsonl"), row)
    if is_run and relevant:
        subprocess.Popen([sys.executable, os.path.abspath(__file__), "sample", logdir, repr(ended), json.dumps(watch_paths(mounts)), tag],
                         start_new_session=True, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    return code


def sample(logdir, t0, paths, tag):
    t0 = float(t0)
    paths = json.loads(paths)
    for offset in SCHEDULE:
        delay = t0 + offset - now()
        if delay > 0:
            time.sleep(delay)
        write_jsonl(os.path.join(logdir, "samples.jsonl"), {"tag": tag, "offset": offset, "t": now(), "seen": [lst(p) for p in paths]})


def watch(root, out):
    state, known, last_scan = {}, set(), 0.0
    skip = {".git", "site-packages", "node_modules", "target", ".venv", ".m2", "lib", "include", "__pycache__", ".gradle"}
    while True:
        tick = now()
        if tick - last_scan > 0.5:
            last_scan = tick
            for directory, subdirs, _files in os.walk(root):
                depth = directory[len(root):].count(os.sep)
                subdirs[:] = [d for d in subdirs if d not in skip and depth < 9]
                if os.path.basename(directory) == ".kriya":
                    known.add(directory)
                    subdirs[:] = []
        for kriya in list(known):
            view = tuple(bool(lst(kriya + rel)["x"]) for rel in ("", "/venv", "/venv/bin", "/venv/bin/python", "/venv/pyvenv.cfg"))
            if state.get(kriya) != view:
                write_jsonl(out, {"t": tick, "kriya": kriya, "dot_kriya_venv_bin_python_cfg": view, "venv": lst(kriya + "/venv")})
                state[kriya] = view
        time.sleep(0.02)


def main():
    cmd = sys.argv[1]
    if cmd == "make-shim":
        directory, real, logdir = sys.argv[2:5]
        os.makedirs(directory, exist_ok=True)
        os.makedirs(logdir, exist_ok=True)
        path = os.path.join(directory, "docker")
        with open(path, "w", encoding="utf-8") as handle:
            handle.write('#!/bin/sh\nexec %s %s shim-run %s %s "$@"\n' % (json.dumps(sys.executable), json.dumps(os.path.abspath(__file__)), json.dumps(real), json.dumps(logdir)))
        os.chmod(path, 0o755)
        print(path)
    elif cmd == "shim-run":
        sys.exit(shim_run(sys.argv[2], sys.argv[3], sys.argv[4:]))
    elif cmd == "sample":
        sample(*sys.argv[2:6])
    elif cmd == "watch":
        watch(sys.argv[2], sys.argv[3])


if __name__ == "__main__":
    main()
